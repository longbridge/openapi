1//! Lenient decoding of the gateway's proto-JSON responses into the `prost`
2//! message types shared with the WebSocket transport.
3//!
4//! The protobuf types derive plain `serde` impls, which expect every field to
5//! be present with its native JSON type. The gateway's proto-JSON differs in a
6//! few ways that this deserializer absorbs, so the existing
7//! `TryFrom<proto>` conversions can be reused unchanged:
8//!
9//! - 64-bit integers (and occasionally other scalars) are encoded as strings
10//! - fields holding their default value may be omitted or `null`
11//! - keys may use the lowerCamelCase JSON name instead of the proto name
12

14    Deserializer,
15    de::{
16        self, DeserializeOwned, DeserializeSeed, IntoDeserializer, MapAccess, SeqAccess, Visitor,
17    },
18    forward_to_deserialize_any,
19};
20use serde_json::{Map, Value};

24/// Recursively remove `null` object members, so an absent `Option<Message>`
25/// is omitted from a request body rather than sent as `null`.
26pub(crate) fn strip_nulls(value: &mut Value) {
27    match value {
28        Value::Object(map) => {
29            map.retain(|_, v| !v.is_null());
30            map.values_mut().for_each(strip_nulls);
31        }
32        Value::Array(arr) => arr.iter_mut().for_each(strip_nulls),
33        _ => {}
34    }
35}

37/// Deserialize `T` from a gateway proto-JSON value.
38pub(crate) fn from_value<T: DeserializeOwned>(value: Value) -> Result<T, Error> {
39    T::deserialize(Lenient(value))
40}

42/// A `serde_json::Value` deserializer that treats `null` as the target type's
43/// default value and accepts numbers / booleans encoded as strings.
44struct Lenient(Value);
45
46fn invalid(value: &Value, exp: &dyn de::Expected) -> Error {
47    de::Error::invalid_type(
48        match value {
49            Value::Null => de::Unexpected::Unit,
50            Value::Bool(b) => de::Unexpected::Bool(*b),
51            Value::Number(_) => de::Unexpected::Other("number"),
52            Value::String(s) => de::Unexpected::Str(s),
53            Value::Array(_) => de::Unexpected::Seq,
54            Value::Object(_) => de::Unexpected::Map,
55        },
56        exp,
57    )
58}

60macro_rules! deserialize_int {
61    ($($method:ident => $visit:ident: $ty:ty),* $(,)?) => {
62        $(
63            fn $method<V: Visitor<'de>>(self, visitor: V) -> Result<V::Value, Error> {
64                let n: $ty = match &self.0 {
65                    Value::Null => 0 as $ty,
66                    Value::Number(n) => n
67                        .to_string()
68                        .parse()
69                        .map_err(|_| invalid(&self.0, &visitor))?,
70                    Value::String(s) if s.is_empty() => 0 as $ty,
71                    Value::String(s) => s.parse().map_err(|_| invalid(&self.0, &visitor))?,
72                    Value::Bool(b) => *b as u8 as $ty,
73                    _ => return Err(invalid(&self.0, &visitor)),
74                };
75                visitor.$visit(n)
76            }
77        )*
78    };
79}

81impl<'de> Deserializer<'de> for Lenient {
82    type Error = Error;
83
84    deserialize_int! {
85        deserialize_i8 => visit_i8: i8,
86        deserialize_i16 => visit_i16: i16,
87        deserialize_i32 => visit_i32: i32,
88        deserialize_i64 => visit_i64: i64,
89        deserialize_u8 => visit_u8: u8,
90        deserialize_u16 => visit_u16: u16,
91        deserialize_u32 => visit_u32: u32,
92        deserialize_u64 => visit_u64: u64,
93        deserialize_f32 => visit_f32: f32,
94        deserialize_f64 => visit_f64: f64,
95    }
96
97    fn deserialize_any<V: Visitor<'de>>(self, visitor: V) -> Result<V::Value, Error> {
98        match self.0 {
99            Value::Array(arr) => visitor.visit_seq(Seq(arr.into_iter())),
100            Value::Object(map) => visitor.visit_map(Fields::new(map, &[])),
101            other => other.deserialize_any(visitor),
102        }
103    }
104
105    fn deserialize_bool<V: Visitor<'de>>(self, visitor: V) -> Result<V::Value, Error> {
106        match &self.0 {
107            Value::Null => visitor.visit_bool(false),
108            Value::Bool(b) => visitor.visit_bool(*b),
109            Value::String(s) if s == "true" || s == "false" => visitor.visit_bool(s == "true"),
110            Value::Number(n) => visitor.visit_bool(n.as_f64() != Some(0.0)),
111            _ => Err(invalid(&self.0, &visitor)),
112        }
113    }
114
115    fn deserialize_str<V: Visitor<'de>>(self, visitor: V) -> Result<V::Value, Error> {
116        self.deserialize_string(visitor)
117    }
118
119    fn deserialize_string<V: Visitor<'de>>(self, visitor: V) -> Result<V::Value, Error> {
120        match self.0 {
121            Value::Null => visitor.visit_string(String::new()),
122            Value::String(s) => visitor.visit_string(s),
123            Value::Number(n) => visitor.visit_string(n.to_string()),
124            Value::Bool(b) => visitor.visit_string(b.to_string()),
125            other => Err(invalid(&other, &visitor)),
126        }
127    }
128
129    fn deserialize_option<V: Visitor<'de>>(self, visitor: V) -> Result<V::Value, Error> {
130        match self.0 {
131            Value::Null => visitor.visit_none(),
132            other => visitor.visit_some(Lenient(other)),
133        }
134    }
135
136    fn deserialize_newtype_struct<V: Visitor<'de>>(
137        self,
138        _name: &'static str,
139        visitor: V,
140    ) -> Result<V::Value, Error> {
141        visitor.visit_newtype_struct(self)
142    }
143
144    fn deserialize_seq<V: Visitor<'de>>(self, visitor: V) -> Result<V::Value, Error> {
145        match self.0 {
146            Value::Null => visitor.visit_seq(Seq(Vec::new().into_iter())),
147            Value::Array(arr) => visitor.visit_seq(Seq(arr.into_iter())),
148            other => Err(invalid(&other, &visitor)),
149        }
150    }
151
152    fn deserialize_tuple<V: Visitor<'de>>(
153        self,
154        _len: usize,
155        visitor: V,
156    ) -> Result<V::Value, Error> {
157        self.deserialize_seq(visitor)
158    }
159
160    fn deserialize_tuple_struct<V: Visitor<'de>>(
161        self,
162        _name: &'static str,
163        _len: usize,
164        visitor: V,
165    ) -> Result<V::Value, Error> {
166        self.deserialize_seq(visitor)
167    }
168
169    fn deserialize_map<V: Visitor<'de>>(self, visitor: V) -> Result<V::Value, Error> {
170        match self.0 {
171            Value::Null => visitor.visit_map(Fields::new(Map::new(), &[])),
172            Value::Object(map) => visitor.visit_map(Fields::new(map, &[])),
173            other => Err(invalid(&other, &visitor)),
174        }
175    }
176
177    fn deserialize_struct<V: Visitor<'de>>(
178        self,
179        _name: &'static str,
180        fields: &'static [&'static str],
181        visitor: V,
182    ) -> Result<V::Value, Error> {
183        match self.0 {
184            Value::Null => visitor.visit_map(Fields::new(Map::new(), fields)),
185            Value::Object(map) => visitor.visit_map(Fields::new(map, fields)),
186            other => Err(invalid(&other, &visitor)),
187        }
188    }
189
190    fn deserialize_enum<V: Visitor<'de>>(
191        self,
192        name: &'static str,
193        variants: &'static [&'static str],
194        visitor: V,
195    ) -> Result<V::Value, Error> {
196        self.0.deserialize_enum(name, variants, visitor)
197    }
198
199    fn deserialize_ignored_any<V: Visitor<'de>>(self, visitor: V) -> Result<V::Value, Error> {
200        visitor.visit_unit()
201    }
202
203    forward_to_deserialize_any! {
204        char bytes byte_buf unit unit_struct identifier
205    }
206}

210impl<'de> SeqAccess<'de> for Seq {
211    type Error = Error;
212
213    fn next_element_seed<T: DeserializeSeed<'de>>(
214        &mut self,
215        seed: T,
216    ) -> Result<Option<T::Value>, Error> {
217        self.0
218            .next()
219            .map(|v| seed.deserialize(Lenient(v)))
220            .transpose()
221    }
222}

224/// Map access over an object's entries. For a struct, keys are normalized to
225/// the expected field names and every expected field that is missing is
226/// yielded as `null`, so it decodes to its default value.
227struct Fields {
228    entries: std::vec::IntoIter<(String, Value)>,
229    value: Option<Value>,
230}
231
232impl Fields {
233    fn new(map: Map<String, Value>, fields: &'static [&'static str]) -> Self {
234        let mut entries: Vec<(String, Value)> = Vec::with_capacity(map.len() + fields.len());
235        // camelCase aliases are applied after all exact names, so a payload
236        // carrying both spellings never yields a duplicate: the proto name
237        // wins.
238        let mut aliases: Vec<(String, Value)> = Vec::new();
239        for (key, value) in map {
240            if fields.is_empty() || fields.contains(&key.as_str()) {
241                entries.push((key, value));
242                continue;
243            }
244            let snake = camel_to_snake(&key);
245            if fields.contains(&snake.as_str()) {
246                aliases.push((snake, value));
247            } else {
248                entries.push((key, value));
249            }
250        }
251        for (key, value) in aliases {
252            if !entries.iter().any(|(k, _)| *k == key) {
253                entries.push((key, value));
254            }
255        }
256        for field in fields {
257            if !entries.iter().any(|(key, _)| key == field) {
258                entries.push(((*field).to_string(), Value::Null));
259            }
260        }
261        Self {
262            entries: entries.into_iter(),
263            value: None,
264        }
265    }
266}
267
268impl<'de> MapAccess<'de> for Fields {
269    type Error = Error;
270
271    fn next_key_seed<K: DeserializeSeed<'de>>(
272        &mut self,
273        seed: K,
274    ) -> Result<Option<K::Value>, Error> {
275        match self.entries.next() {
276            Some((key, value)) => {
277                self.value = Some(value);
278                seed.deserialize(key.into_deserializer()).map(Some)
279            }
280            None => Ok(None),
281        }
282    }
283
284    fn next_value_seed<V: DeserializeSeed<'de>>(&mut self, seed: V) -> Result<V::Value, Error> {
285        seed.deserialize(Lenient(self.value.take().unwrap_or(Value::Null)))
286    }
287}

289fn camel_to_snake(s: &str) -> String {
290    let mut out = String::with_capacity(s.len() + 4);
291    for ch in s.chars() {
292        if ch.is_ascii_uppercase() {
293            out.push('_');
294            out.push(ch.to_ascii_lowercase());
295        } else {
296            out.push(ch);
297        }
298    }
299    out
300}

310    fn decode_quote_with_string_int64_and_missing_fields() {
311        // int64 as string, enum as int, camelCase keys, omitted fields, null
312        // sub-message, and an extra field unknown to the proto.
313        let resp: SecurityQuoteResponse = from_value(json!({
314            "secuQuote": [{
315                "symbol": "700.HK",
316                "lastDone": "432.200",
317                "prev_close": "430.000",
318                "volume": "12345678",
319                "turnover": "5300000000.000",
320                "timestamp": "1700000000",
321                "trade_status": 0,
322                "volume_str": "12345678",
323                "pre_market_quote": null,
324            }]
325        }))
326        .unwrap();
327        let q = &resp.secu_quote[0];
328        assert_eq!(q.symbol, "700.HK");
329        assert_eq!(q.last_done, "432.200");
330        assert_eq!(q.prev_close, "430.000");
331        assert_eq!(q.volume, 12345678);
332        assert_eq!(q.timestamp, 1700000000);
333        assert_eq!(q.trade_status, 0);

352    #[test]
353    fn decode_edge_cases() {
354        use longbridge_proto::quote::{Brokers, StrikePriceInfo};
355        // null inside arrays, mixed string/number ints, bool spellings.
356        let b: Brokers = from_value(json!({
357            "position": "1", "broker_ids": ["1", 2, null, "-3"]
358        }))
359        .unwrap();
360        assert_eq!(b.broker_ids, vec![1, 2, 0, -3]);
361        for (v, want) in [
362            (json!("true"), true),
363            (json!(true), true),
364            (json!(1), true),
365            (json!(null), false),
366        ] {
367            let s: StrikePriceInfo = from_value(json!({ "price": "1", "standard": v })).unwrap();
368            assert_eq!(s.standard, want);
369        }
370        // camelCase and proto name both present: proto name wins, no duplicate.
371        for order in [
372            json!({ "orderNum": 9, "order_num": "3", "position": 1 }),
373            json!({ "order_num": "3", "orderNum": 9, "position": 1 }),
374        ] {
375            let d: SecurityDepthResponse = from_value(json!({ "ask": [order] })).unwrap();
376            assert_eq!(d.ask[0].order_num, 3);
377        }
378        // Values that must not be silently accepted.
379        assert!(
380            from_value::<SecurityDepthResponse>(json!({ "ask": [{ "volume": "1.5" }] })).is_err()
381        );
382        assert!(
383            from_value::<SecurityDepthResponse>(
384                json!({ "ask": [{ "volume": "99999999999999999999" }] })
385            )
386            .is_err()
387        );
388        // Enum names are not accepted (the gateway sends numbers).
389        assert!(
390            from_value::<SecurityQuoteResponse>(json!({
391                "secu_quote": [{ "trade_status": "Normal" }]
392            }))
393            .is_err()
394        );
395    }
