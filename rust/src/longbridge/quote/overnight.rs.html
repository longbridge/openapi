1//! Keeps HTTP-transport quote data aligned with the WebSocket's US overnight
2//! rule.
3//!
4//! This is a client-side stopgap for a gateway gap (no REST opt-out for US
5//! overnight data). It relies on observed server behaviour (a 1000-bar request
6//! cap, date ranges served newest-first, offset queries anchored inclusively
7//! in New York time) and should be removed once the REST endpoints honour
8//! `need_over_night_quote` like the WebSocket does.
9//!
10//! The quote WebSocket only returns the US overnight session when the
11//! connection opted in (`need_over_night_quote`, i.e.
12//! `Config::enable_overnight`), and it applies that *before* count / range
13//! caps. The REST endpoints always return overnight data and offer no opt-out,
14//! so with [`QuoteTransport::Http`](crate::QuoteTransport::Http) and overnight
15//! disabled the SDK drops overnight bars client-side and, for capped
16//! candlestick windows, pages further so both transports yield the same bars.
17

24/// Server-side cap on the `count` of a candlestick request (`count = 5000` is
25/// rejected with `301607`; date-range queries observed returning up to 1440
26/// bars, so a page of at least this size is treated as capped).
27pub(crate) const MAX_HISTORY_CANDLESTICKS: usize = 1000;
28
29/// Hard limit on extra requests one call may issue while topping up a window.
30pub(crate) const MAX_REFILL_ROUNDS: usize = 8;
31
32/// `true` for US equities / ETFs (`AAPL.US`), whose overnight session is what
33/// the WebSocket gates. Indices (`.VIX.US`) trade on their own schedule and
34/// are left alone.
35pub(crate) fn is_us_equity_symbol(symbol: &str) -> bool {
36    let n = symbol.len();
37    n > 3
38        && symbol.is_char_boundary(n - 3)
39        && symbol[n - 3..].eq_ignore_ascii_case(".US")
40        && !symbol.starts_with('.')
41}

43/// Whether `timestamp` falls in the US overnight session (20:00–04:00 New
44/// York).
45pub(crate) fn is_us_overnight(timestamp: OffsetDateTime) -> bool {
46    longbridge_candlesticks::markets::US.trade_session(timestamp)
47        == Some(longbridge_candlesticks::TRADE_SESSION_OVERNIGHT)
48}

50fn is_overnight_candlestick(candlestick: &quote::Candlestick) -> bool {
51    candlestick.trade_session == quote::TradeSession::OvernightTrade as i32
52}

54/// New York wall-clock time of `timestamp`, as used by the offset query.
55pub(crate) fn new_york_local(timestamp: OffsetDateTime) -> PrimitiveDateTime {
56    let local = timestamp.to_timezone(NEW_YORK);
57    PrimitiveDateTime::new(local.date(), local.time())
58}

60/// New York calendar date of `timestamp`.
61pub(crate) fn new_york_date(timestamp: OffsetDateTime) -> Date {
62    timestamp.to_timezone(NEW_YORK).date()
63}

65/// How many bars a candlestick window should hold once overnight bars are
66/// dropped, i.e. what the WebSocket (which drops them before capping) returns.
67#[derive(Debug, Clone, Copy)]
68pub(crate) enum CandlestickWindow {
69    /// A count-limited query (`count` requested).
70    Count(usize),
71    /// A date-range query, served newest-first and capped at a fixed size.
72    DateRange { start: Option<Date> },
73}
74
75/// The number of bars the WebSocket would return for `window` given the raw
76/// page, or `None` if the page is not capped (nothing to top up).
77pub(crate) fn window_target(
78    window: CandlestickWindow,
79    raw: &[quote::Candlestick],
80) -> Option<usize> {
81    match window {
82        // A page shorter than `count` is either all the data there is or the
83        // server's cap; either way it is the window the WebSocket would return.
84        CandlestickWindow::Count(count) => Some(count.min(raw.len())),
85        // A page shorter than the request cap cannot have been cut off. A
86        // full-size page is capped unless it already reaches the first bar of
87        // `start` (midnight New York, where the overnight session begins);
88        // without a `start` it is taken as capped.
89        CandlestickWindow::DateRange { start } => {
90            let capped = raw.len() >= MAX_HISTORY_CANDLESTICKS
91                && match (start, raw_edge(raw, false)) {
92                    (Some(start), Some(earliest)) => new_york_local(earliest) > start.midnight(),
93                    (None, Some(_)) => true,
94                    _ => false,
95                };
96            capped.then_some(raw.len())
97        }
98    }
99}

101/// Earliest (`forward == false`) or latest (`forward == true`) timestamp in a
102/// raw page.
103pub(crate) fn raw_edge(
104    candlesticks: &[quote::Candlestick],
105    forward: bool,
106) -> Option<OffsetDateTime> {
107    let timestamps = candlesticks.iter().map(|candlestick| candlestick.timestamp);
108    if forward {
109        timestamps.max()
110    } else {
111        timestamps.min()
112    }
113    .and_then(|timestamp| OffsetDateTime::from_unix_timestamp(timestamp).ok())
114}

116/// Convert a raw page, dropping overnight candlesticks. Returns the kept
117/// candlesticks and how many were dropped.
118pub(crate) fn drop_overnight(
119    candlesticks: Vec<quote::Candlestick>,
120) -> Result<(Vec<Candlestick>, usize)> {
121    let total = candlesticks.len();
122    let kept = candlesticks
123        .into_iter()
124        .filter(|candlestick| !is_overnight_candlestick(candlestick))
125        .map(TryInto::try_into)
126        .collect::<Result<Vec<_>>>()?;
127    let dropped = total - kept.len();
128    Ok((kept, dropped))
129}

131/// Top up a capped candlestick window after overnight bars were dropped.
132///
133/// `candlesticks` is the already-filtered first page and `cursor` the edge of
134/// its *raw* window (overnight bars included). Pages are fetched beyond the
135/// cursor with `fetch(anchor_in_new_york, count)`, where the server treats the
136/// anchor as inclusive, until `target` bars are collected, the data runs out,
137/// `bound` (a New York date, for date-range queries) is crossed, or
138/// [`MAX_REFILL_ROUNDS`] is reached.
139pub(crate) async fn refill<F, Fut>(
140    forward: bool,
141    mut candlesticks: Vec<Candlestick>,
142    mut cursor: Option<OffsetDateTime>,
143    target: usize,
144    bound: Option<Date>,
145    mut fetch: F,
146) -> Result<(Vec<Candlestick>, bool)>
147where
148    F: FnMut(PrimitiveDateTime, usize) -> Fut,
149    Fut: Future<Output = Result<Vec<quote::Candlestick>>>,
150{
151    let mut stalled = false;
152    let mut round_limit_hit = true;
153    for _ in 0..MAX_REFILL_ROUNDS {
154        if candlesticks.len() >= target {
155            round_limit_hit = false;
156            break;
157        }
158        let Some(anchor) = cursor else {
159            round_limit_hit = false;
160            break;
161        };
162        let need = target - candlesticks.len();
163        // Over-fetch a little for the anchor bar (returned and discarded) and
164        // scattered overnight bars. The overnight session is one contiguous
165        // 20:00–04:00 block, so once a page yields nothing the cursor is
166        // inside it: jump with a full page rather than inching through it.
167        let count = if stalled {
168            MAX_HISTORY_CANDLESTICKS
169        } else {
170            (need + need / 2 + 1).min(MAX_HISTORY_CANDLESTICKS)
171        };
172        let raw = fetch(new_york_local(anchor), count).await?;
173        let exhausted = raw.len() < count;
174        let next_cursor = raw_edge(&raw, forward);
175
176        let (mut more, _) = drop_overnight(raw)?;
177        more.retain(|candlestick| {
178            if forward {
179                candlestick.timestamp > anchor
180            } else {
181                candlestick.timestamp < anchor
182            }
183        });
184        let mut crossed_bound = false;
185        if let Some(bound) = bound {
186            more.retain(|candlestick| {
187                let date = new_york_date(candlestick.timestamp);
188                let inside = if forward {
189                    date <= bound
190                } else {
191                    date >= bound
192                };
193                crossed_bound |= !inside;
194                inside
195            });
196        }
197
198        stalled = more.is_empty();
199        if forward {
200            more.truncate(need);
201            candlesticks.extend(more);
202        } else {
203            let skip = more.len().saturating_sub(need);
204            candlesticks.splice(0..0, more.into_iter().skip(skip));
205        }
206
207        if exhausted || crossed_bound || next_cursor.is_none() || next_cursor == cursor {
208            round_limit_hit = false;
209            break;
210        }
211        cursor = next_cursor;
212    }
213
214    // The flag means "stopped by the round limit while still short".
215    let round_limit_hit = round_limit_hit && candlesticks.len() < target;
216    Ok((candlesticks, round_limit_hit))
217}

228    /// One-minute bars for New York 2026-10-09 (EDT) from `start` for `n`
229    /// minutes, with the trade session derived from the clock.
230    fn timeline(start: OffsetDateTime, n: usize) -> Vec<quote::Candlestick> {
231        (0..n as i64)
232            .map(|i| {
233                let timestamp = start + time::Duration::minutes(i);
234                let local = timestamp.to_timezone(NEW_YORK).time();
235                let session = match local.hour() {
236                    0..=3 | 20..=23 => quote::TradeSession::OvernightTrade,
237                    4..=8 => quote::TradeSession::PreTrade,
238                    9 if local.minute() < 30 => quote::TradeSession::PreTrade,
239                    9..=15 => quote::TradeSession::NormalTrade,
240                    _ => quote::TradeSession::PostTrade,
241                };
242                quote::Candlestick {
243                    close: "1".into(),
244                    open: "1".into(),
245                    low: "1".into(),
246                    high: "1".into(),
247                    volume: 1,
248                    turnover: "1".into(),
249                    timestamp: timestamp.unix_timestamp(),
250                    trade_session: session as i32,
251                }
252            })
253            .collect()
254    }

256    /// A fake server: `count` bars from the anchor (inclusive) in `forward`
257    /// direction, ascending.
258    fn server(
259        data: Vec<quote::Candlestick>,
260        forward: bool,
261        calls: Rc<Cell<usize>>,
262    ) -> impl FnMut(PrimitiveDateTime, usize) -> std::future::Ready<Result<Vec<quote::Candlestick>>>
263    {
264        move |anchor, count| {
265            calls.set(calls.get() + 1);
266            let anchor = anchor
267                .assume_timezone(NEW_YORK)
268                .unwrap_first()
269                .unix_timestamp();
270            let page: Vec<_> = if forward {
271                data.iter()
272                    .filter(|c| c.timestamp >= anchor)
273                    .take(count)
274                    .cloned()
275                    .collect()
276            } else {
277                let mut page: Vec<_> = data
278                    .iter()
279                    .rev()
280                    .filter(|c| c.timestamp <= anchor)
281                    .take(count)
282                    .cloned()
283                    .collect();
284                page.reverse();
285                page
286            };
287            std::future::ready(Ok(page))
288        }
289    }

291    fn minutes(candlesticks: &[Candlestick]) -> Vec<String> {
292        candlesticks
293            .iter()
294            .map(|c| {
295                let local = c.timestamp.to_timezone(NEW_YORK);
296                format!("{:02}:{:02}", local.hour(), local.minute())
297            })
298            .collect()
299    }

535    fn window_targets() {
536        let data = timeline(DAY, 24 * 60);
537        assert_eq!(
538            window_target(CandlestickWindow::Count(1000), &data[..300]),
539            Some(300)
540        );
541        assert_eq!(
542            window_target(CandlestickWindow::Count(100), &data[..300]),
543            Some(100)
544        );
545        let start = Some(DAY.date());
546        // Reaches midnight of `start`: not capped.
547        assert_eq!(
548            window_target(CandlestickWindow::DateRange { start }, &data),
549            None
550        );
551        // A short page is never capped, whatever its first bar.
552        assert_eq!(
553            window_target(CandlestickWindow::DateRange { start }, &data[20 * 60..]),
554            None
555        );
556        // A full-size page starting after midnight of `start`: capped.
557        assert_eq!(
558            window_target(CandlestickWindow::DateRange { start }, &data[4 * 60..]),
559            Some(1200)
560        );
561        // Earlier `start` the page did not reach: capped.
562        let earlier = Some(DAY.date().previous_day().unwrap());
563        assert_eq!(
564            window_target(CandlestickWindow::DateRange { start: earlier }, &data),
565            Some(1440)
566        );
567        // No start: only a full-size page counts as capped.
568        assert_eq!(
569            window_target(CandlestickWindow::DateRange { start: None }, &data[..999]),
570            None
571        );
572        assert_eq!(
573            window_target(CandlestickWindow::DateRange { start: None }, &data),
574            Some(1440)
575        );
576        assert_eq!(
577            window_target(CandlestickWindow::DateRange { start }, &[]),
578            None
579        );
580    }

582    #[test]
583    fn round_limit_is_enforced() {
584        // A pathological server that always returns a full page of overnight
585        // bars.
586        let calls = Rc::new(Cell::new(0));
587        let c2 = calls.clone();
588        let mut t = DAY.unix_timestamp();
589        let fetch = move |_anchor: PrimitiveDateTime, count: usize| {
590            c2.set(c2.get() + 1);
591            t -= 60 * count as i64;
592            let page = (0..count as i64)
593                .map(|i| quote::Candlestick {
594                    close: "1".into(),
595                    open: "1".into(),
596                    low: "1".into(),
597                    high: "1".into(),
598                    volume: 1,
599                    turnover: "1".into(),
600                    timestamp: t + i * 60,
601                    trade_session: quote::TradeSession::OvernightTrade as i32,
602                })
603                .collect();
604            std::future::ready(Ok(page))
605        };
606        let rt = tokio::runtime::Builder::new_current_thread()
607            .build()
608            .unwrap();
609        let out = rt
610            .block_on(refill(false, vec![], Some(DAY), 50, None, fetch))
611            .unwrap();
612        assert!(out.1, "round limit not reported");
613        let out = out.0;
614        assert!(out.is_empty());
615        assert_eq!(calls.get(), MAX_REFILL_ROUNDS);
616    }

619    fn us_overnight_window() {
620        // EDT (2026-10-09) and EST (2026-01-09) boundaries.
621        for day in [
622            datetime!(2026-10-09 00:00 -04:00),
623            datetime!(2026-01-09 00:00 -05:00),
624        ] {
625            let at = |h: i64, m: i64| day + time::Duration::minutes(h * 60 + m);
626            assert!(is_us_overnight(at(0, 0)));
627            assert!(is_us_overnight(at(3, 59)));
628            assert!(!is_us_overnight(at(4, 0)));
629            assert!(!is_us_overnight(at(9, 30)));
630            assert!(!is_us_overnight(at(16, 0)));
631            assert!(!is_us_overnight(at(19, 59)));
632            assert!(is_us_overnight(at(20, 0)));
633            assert!(is_us_overnight(at(23, 59)));
634        }
635        assert!(is_us_equity_symbol("AAPL.US"));
636        assert!(is_us_equity_symbol("aapl.us"));
637        assert!(!is_us_equity_symbol(".US"));
638        assert!(is_us_equity_symbol("AAPL251017C340000.US"));
639        assert!(!is_us_equity_symbol(".VIX.US"));
640        assert!(!is_us_equity_symbol("700.HK"));
