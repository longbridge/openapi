    Interface MessagePayload

    Payload of a message stream event — an incremental text chunk. This is the highest-frequency event; concatenate text fragments in arrival order.

    interface MessagePayload {
        text: string;
        messageType: string;
        key: string;
        startedAt: number;
        stage: string;
        stageTitle: string;
        stageFinishedTitle: string;
        outputs?: any;
    }
